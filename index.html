create-payment-link.js
const onafi = require("@onafi/node")(process.env.ONAFI_KEY);

// Create a payment link for a product
const link = await onafi.paymentLinks.create({
  name: "Design Workshop Ticket",
  amount: 15000,
  currency: "NGN",
  redirect_url: "https://yourstore.com/thanks"
});

console.log(link.url);
